#include <stdio.h>
#include <stdlib.h>

void inputArray(int arr[], int n)
{
    int i;

    printf("Enter %d elements:\n", n);

    for (i = 0; i < n; i++)
    {
        scanf("%d", &arr[i]);
    }
}

void printArray(int arr[], int n)
{
    int i;

    for (i = 0; i < n; i++)
    {
        printf("%d ", arr[i]);
    }

    printf("\n");
}

void merge(int arr[], int left, int mid, int right)
{
    int n1;
    int n2;
    int i;
    int j;
    int k;

    n1 = mid - left + 1;
    n2 = right - mid;

    int *L = (int *)malloc(n1 * sizeof(int));
    int *R = (int *)malloc(n2 * sizeof(int));

    for (i = 0; i < n1; i++)
    {
        L[i] = arr[left + i];
    }

    for (j = 0; j < n2; j++)
    {
        R[j] = arr[mid + 1 + j];
    }

    i = 0;
    j = 0;
    k = left;

    while (i < n1 && j < n2)
    {
        if (L[i] <= R[j])
        {
            arr[k] = L[i];
            i++;
        }
        else
        {
            arr[k] = R[j];
            j++;
        }

        k++;
    }

    while (i < n1)
    {
        arr[k] = L[i];
        i++;
        k++;
    }

    while (j < n2)
    {
        arr[k] = R[j];
        j++;
        k++;
    }

    free(L);
    free(R);
}

void mergeSort(int arr[], int left, int right)
{
    int mid;

    if (left < right)
    {
        mid = left + (right - left) / 2;

        mergeSort(arr, left, mid);
        mergeSort(arr, mid + 1, right);

        merge(arr, left, mid, right);
    }
}

int findMinimum(int arr[], int n)
{
    int i;
    int minimum;

    minimum = arr[0];

    for (i = 1; i < n; i++)
    {
        if (arr[i] < minimum)
        {
            minimum = arr[i];
        }
    }

    return minimum;
}

int findMaximum(int arr[], int n)
{
    int i;
    int maximum;

    maximum = arr[0];

    for (i = 1; i < n; i++)
    {
        if (arr[i] > maximum)
        {
            maximum = arr[i];
        }
    }

    return maximum;
}

int findSum(int arr[], int n)
{
    int i;
    int sum;

    sum = 0;

    for (i = 0; i < n; i++)
    {
        sum = sum + arr[i];
    }

    return sum;
}

double findAverage(int arr[], int n)
{
    int sum;

    sum = findSum(arr, n);

    return (double)sum / n;
}

void reverseArray(int arr[], int n)
{
    int left;
    int right;
    int temp;

    left = 0;
    right = n - 1;

    while (left < right)
    {
        temp = arr[left];
        arr[left] = arr[right];
        arr[right] = temp;

        left++;
        right--;
    }
}

int main()

    int n;
    int choice;
    int *arr;

    printf("Enter number of elements: ");
    scanf("%d", &n);

        printf("Invalid size.\n");
        return 0;
    }

    arr = (int *)malloc(n * sizeof(int));

    if (arr == NULL)
    {
        printf("Memory allocation failed.\n");
        return 1;
    }

    inputArray(arr, n);

    while (1)
    {
        printf("\n");

        printf("2. Sort Array\n");
        printf("3. Find Minimum\n");

        printf("5. Find Sum\n");
        printf("6. Find Average\n");
        printf("7. Reverse Array\n");
        printf("8. Exit\n");
        printf("Enter your choice: ");

        scanf("%d", &choice);

        if (choice == 1)
        {
            printArray(arr, n);
        }
        else if (choice == 2)
        {
            mergeSort(arr, 0, n - 1);

            printf("Array sorted successfully.\n");
            printArray(arr, n);
        }
        else if (choice == 3)
        {
            printf("Minimum = %d\n", findMinimum(arr, n));
        }
        else if (choice == 4)
        {
            printf("Maximum = %d\n", findMaximum(arr, n));
        }
        else if (choice == 5)
        {
            printf("Sum = %d\n", findSum(arr, n));
        }
        else if (choice == 6)
        {
            printf("Average = %.2f\n", findAverage(arr, n));
        }
        else if (choice == 7)
        {
            reverseArray(arr, n);

            printf("Array reversed successfully.\n");

        }
        else if (choice == 8)

        }
        else

    }

}
